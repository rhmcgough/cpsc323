! Find x to the power of y !
function power(x real, y integer, result real) 
{ 
	while (y != 0) 
	{
		if (y < 0) 
		{		
			result = result / x;
			y = y + 1;
		}
		else 
		{
			result = result * x;
			y = y - 1;
		}
		fi
	}

	return result;
}

@
	integer y;
	real x;
	real result;

@
	get (x, y);
	put (power (x, y, result));
